cmake_minimum_required(VERSION 3.20)
project(nwd_reader VERSION 0.13.0 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(Threads REQUIRED)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ZLIB_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SKIP_INSTALL_ALL ON) # zlib is installed through the nwd export below.
add_subdirectory(third_party/zlib-1.3.1 EXCLUDE_FROM_ALL)
set_target_properties(zlibstatic PROPERTIES EXPORT_NAME zlib
  INTERFACE_INCLUDE_DIRECTORIES "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/third_party/zlib-1.3.1>;$<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/third_party/zlib-1.3.1>;$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>")
add_library(nwd_sqlite STATIC third_party/sqlite/sqlite3.c)
set_target_properties(nwd_sqlite PROPERTIES EXPORT_NAME sqlite)
target_compile_definitions(nwd_sqlite PRIVATE SQLITE_OMIT_LOAD_EXTENSION SQLITE_TEMP_STORE=3 SQLITE_DQS=0)
target_link_libraries(nwd_sqlite PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
add_library(nwd_reader STATIC src/container.cpp src/geometry.cpp src/attributes.cpp src/scene.cpp src/metadata.cpp src/viewpoints.cpp src/access.cpp src/nwf.cpp src/project.cpp src/assets.cpp src/schema.cpp src/products.cpp src/blowfish.cpp src/database.cpp src/lightworks.cpp third_party/tiny-aes/aes.c)
set_source_files_properties(third_party/tiny-aes/aes.c PROPERTIES COMPILE_DEFINITIONS "CBC=0;CTR=0;AES_init_ctx=nwd_lwa_aes_init;AES_ECB_decrypt=nwd_lwa_aes_decrypt;AES_ECB_encrypt=nwd_lwa_aes_encrypt")
add_library(nwd::reader ALIAS nwd_reader)
set_target_properties(nwd_reader PROPERTIES EXPORT_NAME reader)
target_compile_features(nwd_reader PUBLIC cxx_std_20)
target_include_directories(nwd_reader PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}> PRIVATE src third_party)
target_link_libraries(nwd_reader PRIVATE zlibstatic nwd_sqlite Threads::Threads)
if(MSVC)
 target_compile_options(nwd_reader PRIVATE /W4 /permissive- /utf-8)
else()
 target_compile_options(nwd_reader PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS nwd_reader zlibstatic nwd_sqlite EXPORT nwdTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES third_party/zlib-1.3.1/LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/nwd/licenses RENAME zlib.txt)
install(FILES third_party/json-LICENSE.MIT DESTINATION ${CMAKE_INSTALL_DATADIR}/nwd/licenses)
install(FILES third_party/blowfish-LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/nwd/licenses)
install(EXPORT nwdTargets NAMESPACE nwd:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/nwd)
configure_package_config_file(cmake/nwdConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/nwdConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/nwd)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/nwdConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/nwdConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/nwdConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/nwd)

install(FILES third_party/sqlite/LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/nwd/licenses RENAME sqlite.txt)

install(FILES third_party/tiny-aes/unlicense.txt DESTINATION ${CMAKE_INSTALL_DATADIR}/nwd/licenses RENAME tiny-aes.txt)
